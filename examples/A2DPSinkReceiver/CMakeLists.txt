cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(A2DPSinkReceiver C CXX ASM)
pico_sdk_init()
set(SINK_PEER_ADDRESS "" CACHE STRING "Exact intended source XX:XX:XX:XX:XX:XX")
if(SINK_PEER_ADDRESS AND NOT SINK_PEER_ADDRESS MATCHES "^[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]$")
  message(FATAL_ERROR "Invalid SINK_PEER_ADDRESS")
endif()
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/picow_a2dp_sink.cmake)
add_executable(a2dp_sink_receiver main.cpp)
picow_target_a2dp_sink(a2dp_sink_receiver ${CMAKE_CURRENT_LIST_DIR})
target_compile_definitions(a2dp_sink_receiver PRIVATE SINK_PEER_ADDRESS="${SINK_PEER_ADDRESS}"
    PICO_STDIO_USB_STDOUT_TIMEOUT_US=0 CFG_TUD_CDC_TX_BUFSIZE=1024)
pico_enable_stdio_usb(a2dp_sink_receiver 1)
pico_enable_stdio_uart(a2dp_sink_receiver 0)
pico_add_extra_outputs(a2dp_sink_receiver)
