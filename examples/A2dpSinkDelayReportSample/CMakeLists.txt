cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(A2dpSinkDelayReportSample C CXX ASM)
pico_sdk_init()

option(SINK_DELAY_ENABLE_RADIO "Compile the manually controlled radio path (OFF by default)" OFF)
set(SINK_DELAY_PEER "" CACHE STRING "Exact intended source address; empty by default")
if(SINK_DELAY_PEER AND NOT SINK_DELAY_PEER MATCHES "^[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]$")
    message(FATAL_ERROR "SINK_DELAY_PEER must be empty or six hexadecimal octets")
endif()
string(TOUPPER "${SINK_DELAY_PEER}" SINK_DELAY_PEER_UPPER)
if(SINK_DELAY_ENABLE_RADIO AND (NOT SINK_DELAY_PEER OR
    SINK_DELAY_PEER_UPPER STREQUAL "00:00:00:00:00:00" OR
    SINK_DELAY_PEER_UPPER STREQUAL "FF:FF:FF:FF:FF:FF"))
    message(FATAL_ERROR "Radio requires an explicit nonzero, nonbroadcast SINK_DELAY_PEER")
endif()
if(SINK_DELAY_PEER)
    string(REPLACE ":" ",0x" SINK_DELAY_PEER_BYTES "${SINK_DELAY_PEER}")
    set(SINK_DELAY_PEER_BYTES "0x${SINK_DELAY_PEER_BYTES}")
else()
    set(SINK_DELAY_PEER_BYTES "0,0,0,0,0,0")
endif()
configure_file(sample_config.h.in sample_config.h)
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/picow_a2dp_sink.cmake)
add_executable(a2dp_sink_delay_report_sample main.cpp)
picow_target_a2dp_sink(a2dp_sink_delay_report_sample ${CMAKE_CURRENT_LIST_DIR})
target_include_directories(a2dp_sink_delay_report_sample PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_options(a2dp_sink_delay_report_sample PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra>)
# Console output can be lost, but must not wait indefinitely for the USB host.
target_compile_definitions(a2dp_sink_delay_report_sample PRIVATE
    PICO_STDIO_USB_STDOUT_TIMEOUT_US=0 CFG_TUD_CDC_TX_BUFSIZE=1024)
pico_enable_stdio_usb(a2dp_sink_delay_report_sample 1)
pico_enable_stdio_uart(a2dp_sink_delay_report_sample 0)
pico_add_extra_outputs(a2dp_sink_delay_report_sample)
