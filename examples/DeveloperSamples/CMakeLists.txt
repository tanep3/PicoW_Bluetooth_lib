cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(DeveloperSamples C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(PICO_CXX_ENABLE_EXCEPTIONS 1)
option(SAMPLE_USB_CONSOLE "USB CDC instead of UART0" ON)
option(SAMPLE_ALLOW_PAIRING "Confirm SSP for the configured peer only" OFF)
pico_sdk_init()
set(SAMPLE_PEER "" CACHE STRING "Exact peer Bluetooth address; empty disables radio startup")
set(SAMPLE_CHANNEL_0 1 CACHE STRING "Peer RFCOMM server channel for route 0")
set(SAMPLE_CHANNEL_1 2 CACHE STRING "Peer RFCOMM server channel for route 1")
if(NOT SAMPLE_PEER STREQUAL "" AND NOT SAMPLE_PEER MATCHES "^[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]$")
  message(FATAL_ERROR "SAMPLE_PEER must be empty or XX:XX:XX:XX:XX:XX")
endif()
foreach(channel SAMPLE_CHANNEL_0 SAMPLE_CHANNEL_1)
  if(NOT "${${channel}}" MATCHES "^([1-9]|[12][0-9]|30)$")
    message(FATAL_ERROR "${channel} must be 1..30")
  endif()
endforeach()
if(SAMPLE_CHANNEL_0 STREQUAL SAMPLE_CHANNEL_1)
  message(FATAL_ERROR "Two routes need distinct RFCOMM channels on the same peer")
endif()
configure_file(sample_config.h.in sample_config.h @ONLY)
set(ROOT ${CMAKE_CURRENT_LIST_DIR}/../..)
include(${ROOT}/cmake/picow_a2dp_source.cmake)
include(${ROOT}/cmake/picow_avrcp_target.cmake)
include(${ROOT}/cmake/picow_spp.cmake)
foreach(name a2dp_source avrcp_target spp_client spp_two_routes a2dp_avrcp spp_reconnect a2dp_spp_avrcp)
  add_executable(sample_${name} ${name}.cpp)
  target_include_directories(sample_${name} PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  target_compile_options(sample_${name} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra>)
  if(name MATCHES "a2dp")
    picow_target_a2dp_source(sample_${name} ${CMAKE_CURRENT_LIST_DIR})
  endif()
  if(name MATCHES "avrcp")
    picow_target_avrcp_target(sample_${name} ${CMAKE_CURRENT_LIST_DIR})
  endif()
  if(name MATCHES "spp")
    picow_target_spp(sample_${name} ${CMAKE_CURRENT_LIST_DIR})
  endif()
  if(SAMPLE_USB_CONSOLE)
    pico_enable_stdio_usb(sample_${name} 1)
    pico_enable_stdio_uart(sample_${name} 0)
    # Keep diagnostic bursts within a fixed buffer without blocking the radio loop.
    target_compile_definitions(sample_${name} PRIVATE
      PICO_STDIO_USB_STDOUT_TIMEOUT_US=0 CFG_TUD_CDC_TX_BUFSIZE=1024)
  else()
    pico_enable_stdio_usb(sample_${name} 0)
    pico_enable_stdio_uart(sample_${name} 1)
  endif()
  pico_add_extra_outputs(sample_${name})
endforeach()
