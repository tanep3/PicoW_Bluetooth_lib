cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(SppControlMainLoopSample C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
option(SPP_SAMPLE_ENABLE_RADIO "Enable real radio startup and serial-port controls" OFF)
option(SPP_SAMPLE_ENABLE_ERTM "Compile optional RFCOMM ERTM and static storage" OFF)
set(SPP_SAMPLE_FIRST_PEER "00:00:00:00:00:00" CACHE STRING "Authorized first peer Bluetooth address")
set(SPP_SAMPLE_NEXT_PEER "00:00:00:00:00:00" CACHE STRING "Authorized next peer Bluetooth address")
set(SPP_SAMPLE_REMOTE_CHANNEL "1" CACHE STRING "Known first-peer RFCOMM channel (1..30)")
set(SPP_SAMPLE_LOCAL_CHANNEL "2" CACHE STRING "Local RFCOMM listener channel (1..30)")
foreach(peer FIRST NEXT)
  if(NOT SPP_SAMPLE_${peer}_PEER MATCHES "^[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]$")
    message(FATAL_ERROR "SPP_SAMPLE_${peer}_PEER must contain six hexadecimal octets")
  endif()
  if(SPP_SAMPLE_ENABLE_RADIO AND SPP_SAMPLE_${peer}_PEER STREQUAL "00:00:00:00:00:00")
    message(FATAL_ERROR "Radio requires explicit nonzero first and next peer addresses")
  endif()
  string(REPLACE ":" ",0x" SPP_SAMPLE_${peer}_PEER_BYTES "${SPP_SAMPLE_${peer}_PEER}")
  set(SPP_SAMPLE_${peer}_PEER_BYTES "0x${SPP_SAMPLE_${peer}_PEER_BYTES}")
endforeach()
string(TOUPPER "${SPP_SAMPLE_FIRST_PEER}" first_peer_upper)
string(TOUPPER "${SPP_SAMPLE_NEXT_PEER}" next_peer_upper)
if(SPP_SAMPLE_ENABLE_RADIO AND first_peer_upper STREQUAL next_peer_upper)
  message(FATAL_ERROR "Select two different authorized peers")
endif()
foreach(channel REMOTE LOCAL)
  if(NOT SPP_SAMPLE_${channel}_CHANNEL MATCHES "^([1-9]|[12][0-9]|30)$")
    message(FATAL_ERROR "SPP_SAMPLE_${channel}_CHANNEL must be 1..30")
  endif()
endforeach()
configure_file(sample_config.h.in sample_config.h)
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/picow_spp.cmake)
add_executable(spp_control_mainloop_sample main.cpp)
picow_target_spp(spp_control_mainloop_sample ${CMAKE_CURRENT_LIST_DIR})
target_include_directories(spp_control_mainloop_sample PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_options(spp_control_mainloop_sample PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra>)
pico_enable_stdio_usb(spp_control_mainloop_sample 1)
pico_enable_stdio_uart(spp_control_mainloop_sample 0)
pico_add_extra_outputs(spp_control_mainloop_sample)
