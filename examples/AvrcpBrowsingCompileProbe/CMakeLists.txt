cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(AvrcpBrowsingCompileProbe C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(PICO_CXX_ENABLE_EXCEPTIONS 1)
pico_sdk_init()
set(ROOT ${CMAKE_CURRENT_LIST_DIR}/../..)
include(${ROOT}/cmake/picow_a2dp_source.cmake)
include(${ROOT}/cmake/picow_spp.cmake)
include(${ROOT}/cmake/picow_avrcp_browsing.cmake)
add_executable(avrcp_browsing_compile_probe main.cpp)
picow_target_a2dp_source(avrcp_browsing_compile_probe ${CMAKE_CURRENT_LIST_DIR})
picow_target_spp(avrcp_browsing_compile_probe ${CMAKE_CURRENT_LIST_DIR})
picow_target_avrcp_browsing(avrcp_browsing_compile_probe ${CMAKE_CURRENT_LIST_DIR})
target_compile_options(avrcp_browsing_compile_probe PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra>)
pico_add_extra_outputs(avrcp_browsing_compile_probe)
