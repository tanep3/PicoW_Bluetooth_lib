cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(SppFramedDuplexSample C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
option(SPP_SAMPLE_ENABLE_RADIO "Enable manual radio example" OFF)
set(SPP_SAMPLE_PEER "00:00:00:00:00:00" CACHE STRING "Authorized peer address")
set(SPP_SAMPLE_CONTROL_CHANNEL "1" CACHE STRING "Control channel")
set(SPP_SAMPLE_DATA_CHANNEL "2" CACHE STRING "Data channel")
if(NOT SPP_SAMPLE_PEER MATCHES "^[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]$")
  message(FATAL_ERROR "Peer must contain six hexadecimal octets")
endif()
if(SPP_SAMPLE_ENABLE_RADIO AND SPP_SAMPLE_PEER STREQUAL "00:00:00:00:00:00")
  message(FATAL_ERROR "Radio requires explicit nonzero peer")
endif()
foreach(role CONTROL DATA)
  if(NOT SPP_SAMPLE_${role}_CHANNEL MATCHES "^([1-9]|[12][0-9]|30)$")
    message(FATAL_ERROR "Channels must be 1..30")
  endif()
endforeach()
if(SPP_SAMPLE_CONTROL_CHANNEL STREQUAL SPP_SAMPLE_DATA_CHANNEL)
  message(FATAL_ERROR "Control and data require distinct channels")
endif()
string(REPLACE ":" ",0x" SPP_SAMPLE_PEER_BYTES "${SPP_SAMPLE_PEER}")
set(SPP_SAMPLE_PEER_BYTES "0x${SPP_SAMPLE_PEER_BYTES}")
configure_file(sample_config.h.in sample_config.h)
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/picow_spp.cmake)
add_executable(spp_framed_duplex_sample main.cpp)
picow_target_spp(spp_framed_duplex_sample ${CMAKE_CURRENT_LIST_DIR})
target_include_directories(spp_framed_duplex_sample PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_options(spp_framed_duplex_sample PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra>)
target_compile_definitions(spp_framed_duplex_sample PRIVATE PICO_STDIO_USB_STDOUT_TIMEOUT_US=0 PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0 PICO_STDIO_DEADLOCK_TIMEOUT_MS=0)
pico_enable_stdio_usb(spp_framed_duplex_sample 1)
pico_enable_stdio_uart(spp_framed_duplex_sample 0)
pico_add_extra_outputs(spp_framed_duplex_sample)
