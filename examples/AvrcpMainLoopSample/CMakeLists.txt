cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(AvrcpMainLoopSample C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(PICO_CXX_ENABLE_EXCEPTIONS 1)
pico_sdk_init()
option(AVRCP_SAMPLE_ENABLE_RADIO "Enable real radio startup; requires an explicit peer" OFF)
option(AVRCP_SAMPLE_ENABLE_ART_INPUT "Enable explicit one-shot Cover Art console commands" OFF)
if(AVRCP_SAMPLE_ENABLE_ART_INPUT AND NOT AVRCP_SAMPLE_ENABLE_RADIO)
  message(FATAL_ERROR "Cover Art input requires AVRCP_SAMPLE_ENABLE_RADIO=ON")
endif()
set(AVRCP_SAMPLE_PEER "00:00:00:00:00:00" CACHE STRING "Authorized peer Bluetooth address")
if(NOT AVRCP_SAMPLE_PEER MATCHES "^[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]$")
  message(FATAL_ERROR "AVRCP_SAMPLE_PEER must contain six hexadecimal octets")
endif()
if(AVRCP_SAMPLE_ENABLE_RADIO AND AVRCP_SAMPLE_PEER STREQUAL "00:00:00:00:00:00")
  message(FATAL_ERROR "Radio requires an explicit nonzero AVRCP_SAMPLE_PEER")
endif()
string(REPLACE ":" ",0x" AVRCP_SAMPLE_PEER_BYTES "${AVRCP_SAMPLE_PEER}")
set(AVRCP_SAMPLE_PEER_BYTES "0x${AVRCP_SAMPLE_PEER_BYTES}")
configure_file(sample_config.h.in sample_config.h)
set(ROOT ${CMAKE_CURRENT_LIST_DIR}/../..)
include(${ROOT}/cmake/picow_avrcp_browsing.cmake)
include(${ROOT}/cmake/picow_avrcp_cover_art.cmake)
add_executable(avrcp_mainloop_sample main.cpp)
picow_target_avrcp_browsing(avrcp_mainloop_sample ${CMAKE_CURRENT_LIST_DIR})
picow_target_avrcp_cover_art(avrcp_mainloop_sample ${CMAKE_CURRENT_LIST_DIR})
target_include_directories(avrcp_mainloop_sample PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_options(avrcp_mainloop_sample PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra>)
# Do not let a missing/slow console block the polling loop. Logs may be lost.
target_compile_definitions(avrcp_mainloop_sample PRIVATE PICO_STDIO_USB_STDOUT_TIMEOUT_US=0)
pico_enable_stdio_usb(avrcp_mainloop_sample 1)
pico_enable_stdio_uart(avrcp_mainloop_sample 0)
pico_add_extra_outputs(avrcp_mainloop_sample)
