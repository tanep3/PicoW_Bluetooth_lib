cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(AvrcpMediaControlSample C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
option(AVRCP_MEDIA_ENABLE_RADIO "Enable the explicit-start standalone radio console" OFF)
option(AVRCP_MEDIA_ALLOW_PAIRING "Permit explicit fixed-peer five-minute SSP windows" OFF)
set(AVRCP_MEDIA_PEER "00:00:00:00:00:00" CACHE STRING "Exact authorized media-source Bluetooth address")
if(NOT AVRCP_MEDIA_PEER MATCHES "^[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]:[0-9A-Fa-f][0-9A-Fa-f]$")
  message(FATAL_ERROR "AVRCP_MEDIA_PEER requires six hexadecimal octets")
endif()
if(AVRCP_MEDIA_ENABLE_RADIO AND AVRCP_MEDIA_PEER STREQUAL "00:00:00:00:00:00")
  message(FATAL_ERROR "Radio requires an explicit nonzero AVRCP_MEDIA_PEER")
endif()
if(AVRCP_MEDIA_ALLOW_PAIRING AND NOT AVRCP_MEDIA_ENABLE_RADIO)
  message(FATAL_ERROR "Pairing requires the explicit radio build")
endif()
string(REPLACE ":" ",0x" AVRCP_MEDIA_PEER_BYTES "${AVRCP_MEDIA_PEER}")
set(AVRCP_MEDIA_PEER_BYTES "0x${AVRCP_MEDIA_PEER_BYTES}")
configure_file(sample_config.h.in sample_config.h)
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/picow_a2dp_sink.cmake)
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/picow_avrcp_controller.cmake)
add_executable(avrcp_media_control_sample main.cpp)
picow_target_a2dp_sink(avrcp_media_control_sample ${CMAKE_CURRENT_LIST_DIR})
picow_target_avrcp_controller(avrcp_media_control_sample ${CMAKE_CURRENT_LIST_DIR})
target_include_directories(avrcp_media_control_sample PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(avrcp_media_control_sample PRIVATE hardware_watchdog)
target_compile_options(avrcp_media_control_sample PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Werror>)
target_compile_definitions(avrcp_media_control_sample PRIVATE
  PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0 PICO_STDIO_USB_STDOUT_TIMEOUT_US=0
  PICO_STDIO_USB_ENABLE_RESET_VIA_BAUD_RATE=0 CFG_TUD_CDC_TX_BUFSIZE=2048
  USBD_PRODUCT="AVRCP Media Control")
pico_enable_stdio_usb(avrcp_media_control_sample 1)
pico_enable_stdio_uart(avrcp_media_control_sample 0)
pico_add_extra_outputs(avrcp_media_control_sample)
